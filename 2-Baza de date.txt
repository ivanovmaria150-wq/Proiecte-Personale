--1. Tabelul Utilizatori
CREATE TABLE users (
    id SERIAL PRIMARY KEY,
    nume VARCHAR(100) NOT NULL,
    email VARCHAR(100) UNIQUE NOT NULL,
    parola_hash VARCHAR(255) NOT NULL, -- Securitate: stocăm doar hash-ul parolei
    rol VARCHAR(20) CHECK (rol IN ('client', 'fotograf', 'admin')) NOT NULL
);

-- 2. Tabelul Servicii / Pachete Foto
CREATE TABLE services (
    id SERIAL PRIMARY KEY,
    nume_serviciu VARCHAR(100) NOT NULL,
    pret DECIMAL(10, 2) NOT NULL,
    durata_minute INT NOT NULL,
    descriere TEXT,
    activ BOOLEAN DEFAULT TRUE -- Soft delete: dezactivăm serviciul în loc să-l ștergem, pentru a păstra istoricul
);

-- 3. Tabelul Clienți
CREATE TABLE clients (
    id SERIAL PRIMARY KEY,
    user_id INT UNIQUE NOT NULL REFERENCES users(id) ON DELETE CASCADE,
    telefon VARCHAR(20)
);

-- 4. Tabelul Fotografi
CREATE TABLE photographers (
    id SERIAL PRIMARY KEY,
    user_id INT UNIQUE NOT NULL REFERENCES users(id) ON DELETE CASCADE,
    specializare VARCHAR(100)
);

-- 5. Tabelul Programări (Bookings)
CREATE TABLE bookings (
    id SERIAL PRIMARY KEY,
    client_id INT NOT NULL REFERENCES clients(id) ON DELETE CASCADE,
    service_id INT NOT NULL REFERENCES services(id) ON DELETE RESTRICT, -- Nu permitem ștergerea unui serviciu dacă are programări
    photographer_id INT REFERENCES photographers(id) ON DELETE SET NULL,
    data_ora TIMESTAMP NOT NULL,
    status VARCHAR(30) DEFAULT 'in_asteptare',
    CONSTRAINT chk_booking_status CHECK (status IN ('in_asteptare', 'confirmat', 'finalizat', 'anulat'))
);

-- 6. Tabelul Recenzii (Reviews)
CREATE TABLE reviews (
    id SERIAL PRIMARY KEY,
    booking_id INT UNIQUE NOT NULL REFERENCES bookings(id) ON DELETE CASCADE, -- O singură recenzie per programare
    rating INT NOT NULL CHECK (rating BETWEEN 1 AND 5),
    comentariu TEXT
);

-- 7. Tabelul Portofoliu (Galeria de pe site)
CREATE TABLE portfolio (
    id SERIAL PRIMARY KEY,
    titlu VARCHAR(100) NOT NULL,
    categorie VARCHAR(50) NOT NULL, 
    url_imagine VARCHAR(255) NOT NULL,
    descriere TEXT
);
-- Indecsi performanta
CREATE INDEX idx_bookings_client ON bookings (client_id);
CREATE INDEX idx_bookings_photographer_date ON bookings (photographer_id, data_ora);
CREATE INDEX idx_bookings_date ON bookings (data_ora);
